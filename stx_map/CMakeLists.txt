cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(stx_map VERSION 1.0.1 LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 20)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_CXX_EXTENSIONS OFF)
endif()

include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    set(STX_MAP_TESTS_DEFAULT ON)
else()
    set(STX_MAP_TESTS_DEFAULT OFF)
endif()
option(STX_MAP_TESTS "Build the map decoder's tests" ${STX_MAP_TESTS_DEFAULT})

# No dependencies. src/heading_table.inc and src/world_map_data.inc are
# generated by tools/bake (README.md, "Rebaking") and committed.
add_library(stx_map STATIC src/decode.cpp src/levels.cpp src/index.cpp src/projection.cpp src/embedded.cpp)
add_library(stx_map::stx_map ALIAS stx_map)
target_compile_features(stx_map PUBLIC cxx_std_20)
set_target_properties(stx_map PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
target_include_directories(stx_map PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
# Decoding is integer only; the projections are the one floating-point
# computation, kept free of fused multiply-add so every architecture gives
# the same doubles.
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(stx_map PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -ffp-contract=off)
elseif(MSVC)
    target_compile_options(stx_map PRIVATE /W4 /fp:precise)
endif()

if(STX_MAP_TESTS)
    enable_testing()
    add_executable(stx_map_tests tests/map.cpp)
    target_link_libraries(stx_map_tests PRIVATE stx_map::stx_map)
    target_compile_definitions(stx_map_tests PRIVATE STX_MAP_TEST_DATA="${CMAKE_CURRENT_SOURCE_DIR}/tests/data")
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        target_compile_options(stx_map_tests PRIVATE -Wall -Wextra -Wshadow -ffp-contract=off)
    endif()
    add_test(NAME stx_map_tests COMMAND stx_map_tests)
    add_executable(stx_map_decode_fuzz_tests fuzz/map_decode_fuzz.cpp)
    target_link_libraries(stx_map_decode_fuzz_tests PRIVATE stx_map::stx_map)
    target_compile_definitions(stx_map_decode_fuzz_tests PRIVATE STX_MAP_FUZZ_STANDALONE=1)
    add_test(NAME stx_map_decode_fuzz_tests COMMAND stx_map_decode_fuzz_tests)
endif()

install(TARGETS stx_map EXPORT stx_mapTargets
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT stx_mapTargets NAMESPACE stx_map::
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/stx_map)
configure_package_config_file(cmake/stx_mapConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/stx_mapConfig.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/stx_map)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/stx_mapConfigVersion.cmake
    VERSION 1.0.1 COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/stx_mapConfig.cmake
    ${CMAKE_CURRENT_BINARY_DIR}/stx_mapConfigVersion.cmake
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/stx_map)
install(FILES LICENSE PROVENANCE.md DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/stx_map)
install(FILES data/world_map.tmap tools/bake/tmap_decode.js DESTINATION ${CMAKE_INSTALL_DATADIR}/stx_map)
