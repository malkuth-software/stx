cmake_minimum_required(VERSION 3.24)
# stx_skia builds on its own or as the GUI.Forms subdirectory stx_skia/. As a subdirectory it
# declares no project, follows GUI_FORMS_BUILD_TESTS and installs nothing unless asked.
set(STX_SKIA_VERSION 0.1.0)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(stx_skia VERSION ${STX_SKIA_VERSION} LANGUAGES C CXX)
    include(CTest)
    set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
    set(CMAKE_CXX_STANDARD 20)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_CXX_EXTENSIONS OFF)
    set(stx_skia_top_level ON)
    set(stx_skia_tests_default ${BUILD_TESTING})
else()
    set(stx_skia_top_level OFF)
    set(stx_skia_tests_default ${GUI_FORMS_BUILD_TESTS})
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(cmake/sources.cmake)
option(STX_SKIA_TESTS "Build the stx_skia contract and corpus tests" ${stx_skia_tests_default})
option(STX_SKIA_INSTALL "Install the stx_skia package" ${stx_skia_top_level})
option(STX_SKIA_BASELINE "Build the reference profile from STX_SKIA_REFERENCE_ROOT" OFF)
option(STX_SKIA_SANITIZE "Instrument the complete library and tests" OFF)
option(STX_SKIA_SWIZZLE_ONLY "Build only the red/blue swap kernel, stx_skia::swizzle" OFF)
set(STX_SKIA_REFERENCE_ROOT "" CACHE PATH
    "Checkout of malkuth-software/stx_skia at the reference pin, for STX_SKIA_BASELINE")
# The reference profile compiles the library (baseline/, skia/, src/ and include/) from the
# pinned checkout and the tests from this tree, so later changes here are compared against
# the intake as it was pinned. cmake/sources.cmake and cmake/reference.cmake describe that pin
# and stay unchanged; the candidate's own edits to the source list live in cmake/prune.cmake.
if(STX_SKIA_BASELINE)
    if(NOT EXISTS "${STX_SKIA_REFERENCE_ROOT}/baseline/inventory.json")
        message(FATAL_ERROR "STX_SKIA_BASELINE needs STX_SKIA_REFERENCE_ROOT to name a checkout "
            "of malkuth-software/stx_skia at the pin recorded in PROVENANCE.md")
    endif()
    set(stx_library_root "${STX_SKIA_REFERENCE_ROOT}")
else()
    set(stx_library_root "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
set(stx_sources ${STX_SKIA_BASELINE_SOURCES})
if(NOT APPLE)
    list(FILTER stx_sources EXCLUDE REGEX "skia/src/(ports/SkImageGeneratorCG|utils/mac/).*\\.cpp$")
endif()
if(WIN32)
    list(REMOVE_ITEM stx_sources skia/src/ports/SkOSFile_posix.cpp)
    list(APPEND stx_sources skia/src/ports/SkOSFile_win.cpp skia/src/ports/SkLog_win.cpp)
endif()
if(STX_SKIA_BASELINE AND CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64|amd64")
    list(APPEND stx_sources baseline/skia/src/opts/SkOpts_ml3.cpp)
endif()
if(STX_SKIA_BASELINE)
    list(REMOVE_ITEM stx_sources skia/src/core/SkData.cpp)
    list(APPEND stx_sources baseline/SkData.cpp)
    include(cmake/reference.cmake)
else()
    include(cmake/prune.cmake)
endif()
list(TRANSFORM stx_sources PREPEND "${stx_library_root}/")
# The red/blue swap kernel links no Skia code and sees only stx_skia's own headers, so a
# Skia include in it fails to compile and a host can link it on its own.
add_library(stx_skia_swizzle STATIC ${stx_library_root}/src/swizzle.cpp)
add_library(stx_skia::swizzle ALIAS stx_skia_swizzle)
set_target_properties(stx_skia_swizzle PROPERTIES EXPORT_NAME swizzle
    POSITION_INDEPENDENT_CODE ON)
target_compile_features(stx_skia_swizzle PUBLIC cxx_std_20)
target_include_directories(stx_skia_swizzle PUBLIC
    $<BUILD_INTERFACE:${stx_library_root}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(stx_skia_swizzle PRIVATE -fno-exceptions -fno-rtti)
endif()
if(STX_SKIA_SANITIZE)
    target_compile_options(stx_skia_swizzle PUBLIC -fsanitize=address,undefined
        -fno-sanitize=vptr -fno-omit-frame-pointer)
    target_link_options(stx_skia_swizzle PUBLIC -fsanitize=address,undefined)
endif()
# A host that draws without Skia (GUI.Forms with GUI_FORMS_ENABLE_SKIA=OFF) needs only the
# kernel: nothing else is defined, tested or installed.
if(STX_SKIA_SWIZZLE_ONLY)
    return()
endif()
add_library(stx_skia STATIC ${stx_sources} ${stx_library_root}/src/pixel_storage.cpp)
add_library(stx_skia::stx_skia ALIAS stx_skia)
target_link_libraries(stx_skia PUBLIC stx_skia_swizzle)
target_compile_features(stx_skia PUBLIC cxx_std_20)
if(STX_SKIA_BASELINE)
    target_include_directories(stx_skia BEFORE PUBLIC
        $<BUILD_INTERFACE:${stx_library_root}/baseline/skia>)
endif()
target_include_directories(stx_skia PUBLIC
    $<BUILD_INTERFACE:${stx_library_root}/skia>
    $<BUILD_INTERFACE:${stx_library_root}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/stx_skia/skia>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_compile_definitions(stx_skia PUBLIC SK_DISABLE_TRACING SK_GAMMA_APPLY_TO_A8)
target_compile_definitions(stx_skia PRIVATE SKIA_IMPLEMENTATION=1 SK_ENABLE_PRECOMPILE
    SKCMS_DISABLE_HSW SKCMS_DISABLE_SKX)
if(NOT STX_SKIA_BASELINE)
    target_compile_definitions(stx_skia PUBLIC SK_DISABLE_EFFECT_DESERIALIZATION)
    target_compile_definitions(stx_skia PRIVATE STX_SKIA_BASELINE_SIMD)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(stx_skia PRIVATE -ffp-contract=off -fno-exceptions -fno-rtti
        -fvisibility=hidden -fvisibility-inlines-hidden -Wno-attributes)
    if(WIN32 AND CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        target_compile_options(stx_skia PUBLIC -fms-extensions)
    endif()
endif()
set_target_properties(stx_skia PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(APPLE)
    target_link_libraries(stx_skia PUBLIC "-framework CoreFoundation" "-framework CoreGraphics")
    if(STX_SKIA_BASELINE)
        target_link_libraries(stx_skia PUBLIC "-framework ApplicationServices")
    endif()
endif()
if(STX_SKIA_SANITIZE)
    # Skia supplies no RTTI, so vptr instrumentation cannot cross this boundary.
    # Do not disable consumer RTTI: GUI.Forms uses it for its own adapter types.
    target_compile_options(stx_skia PUBLIC -fsanitize=address,undefined
        -fno-sanitize=vptr -fno-omit-frame-pointer)
    target_link_options(stx_skia PUBLIC -fsanitize=address,undefined)
endif()
